            let currentLevel = getCurrentLevel();
            updateFlamesAndCountdown(currentLevel);

            gameState.currentPlayer = (Math.random() < 0.5) ? gameState.man : gameState.woman;
            let targetPlayer = (gameState.currentPlayer === gameState.man) ? gameState.woman : gameState.man;
            
            document.getElementById('current-player-name').innerText = gameState.currentPlayer;

            let questionText = fetchQuestionForLevel(currentLevel, targetPlayer);
            document.getElementById('question-text').innerHTML = questionText;

            resetTimerDisplay();
        }

        function updateFlamesAndCountdown(level) {
            let flames = document.querySelectorAll('.flame');
            flames.forEach((flame, index) => {
                if (index < level) flame.classList.add('active');
                else flame.classList.remove('active');
            });

            let turnsPerLevel = Math.ceil(gameState.totalTurns / 4);
            let currentLevelTurn = ((gameState.currentTurn - 1) % turnsPerLevel) + 1;
            let turnsRemaining = turnsPerLevel - currentLevelTurn + 1;

            let countdownText = document.getElementById('level-countdown-text');
            if (level < 4) {
                countdownText.innerText = `Niveau ${level} — ${turnsRemaining} tour${turnsRemaining > 1 ? 's' : ''} avant le niveau suivant`;
            } else {
                countdownText.innerText = `Niveau 4 — Sommet de l'intensité 🔥`;
            }
        }

        function fetchQuestionForLevel(level, target) {
            let useObject = false;
            if (gameState.activeObjects.length > 0) {
                let chance = 0.2;
                if (gameState.objectFrequency === 'peu') chance = 0.1;
                if (gameState.objectFrequency === 'beaucoup') chance = 0.4;
                if (Math.random() < chance) useObject = true;
            }

            let q = "";
            if (useObject) {
                let randomObj = gameState.activeObjects[Math.floor(Math.random() * gameState.activeObjects.length)];
                let objQuestions = objectDatabase[randomObj];
                q = objQuestions[Math.floor(Math.random() * objQuestions.length)];
                q = `<div class="special-title">⚡ Action avec objet</div>` + q;
            } else {
                let levelPool = questionDatabase[level];
                let available = levelPool.filter(item => !gameState.usedQuestions.has(item));
                if (available.length === 0) available = levelPool;
                
                q = available[Math.floor(Math.random() * available.length)];
                gameState.usedQuestions.add(q);
            }

            return q.replace(/\[target\]/g, target);
        }

        let currentTimerInterval = null;
        function startTimer() {
            window.targetSeconds = 60;
            let display = document.getElementById('timer-countdown');
            let btn = document.getElementById('timer-btn');
            btn.disabled = true;

            if (currentTimerInterval) clearInterval(currentTimerInterval);

            currentTimerInterval = setInterval(() => {
                window.targetSeconds--;
                display.innerText = formatTime(window.targetSeconds);

                if (window.targetSeconds <= 0) {
                    clearInterval(currentTimerInterval);
                    playBipSound();
                    display.innerText = "FIN ! 🔔";
                    btn.disabled = false;
                }
            }, 1000);
        }

        function formatTime(totalSeconds) {
            let minutes = Math.floor(totalSeconds / 60);
            let seconds = totalSeconds % 60;
            return `${minutes}:${seconds < 10 ? '0' : ''}${seconds}`;
        }

        function resetTimerDisplay() {
            if (currentTimerInterval) clearInterval(currentTimerInterval);
            document.getElementById('timer-countdown').innerText = "1:00";
            document.getElementById('timer-btn').disabled = false;
        }

        function skipQuestion() {

            nextTurn();
        }

                </p>
                <button class="primary-btn" onclick="location.reload()">Recommencer une partie</button>
            `;
        }

        function playBipSound() {
            try {
                let audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                let oscillator = audioCtx.createOscillator();
                let gainNode = audioCtx.createGain();
                
                oscillator.type = 'sine';
                oscillator.frequency.value = 587.33;
                gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);
                
                oscillator.connect(gainNode);
                gainNode.connect(audioCtx.destination);
                
                oscillator.start();
                oscillator.stop(audioCtx.currentTime + 0.5);
            } catch (e) {
                console.log("Audio non supporté");
            }
        }
